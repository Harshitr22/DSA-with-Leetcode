      System.out.println("Remove Element: ");

            if(i != arrayRE.length-1) System.out.print(", ");
        }
        System.out.println("}");
        int val = 1;
        System.out.println("The value to be removed from teh array are: " + val);

    public static int removeElement(int[] nums, int val){
        int i = 0;
        for(int j = 0; j < nums.length; j++){
            if(nums[j] != val){
                nums[i] = nums[j];
                i++;
            }
        }
        return i;
    }
