import java.util.*;

public class Main {
    public static void main(String[] args){


      System.out.println("Removing the duplicate from the array: ");
        System.out.print("Before removing duplicates:  ");
        ArrayList<Integer> arrayRD = new ArrayList<>(List.of(0, 0, 1, 1, 1, 2, 2, 3, 3, 4));
        System.out.print("{ ");
        for(int i = 0; i< arrayRD.size(); i++){
            System.out.print(arrayRD.get(i) + " ");
            if(i != arrayRD.size()-1) System.out.print(", ");
        }
        System.out.println("}");
        removeDuplicates(arrayRD);
        System.out.print("After removing duplicates:  ");
        System.out.print("{ ");
        for(int i = 0; i< arrayRD.size(); i++){
            System.out.print(arrayRD.get(i) + " ");
            if(i != arrayRD.size()-1) System.out.print(", ");
        }
        System.out.println("}");
        System.out.println();
        System.out.println("----------------------------------");
        System.out.println();

    }

    public static void removeDuplicates(ArrayList<Integer> array){
        int i = 0;
        int j = 1;
        while(j < array.size()){
            if(array.get(i).equals(array.get(j))){
                array.remove(j);
            }else{
                i++;
                j++;
            }
        }
    }
}
