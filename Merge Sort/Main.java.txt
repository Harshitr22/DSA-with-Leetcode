import java.util.*;

public class Main {
    public static void main(String[] args){


        System.out.println("Merge Sort: ");
        int[] arrayMS = {4,2,6,5,1,3};
        System.out.println("Before Sorting: ");
        System.out.print("{ ");
        for(int i = 0; i<arrayMS.length; i++){
            System.out.print(arrayMS[i] + " ");
            if(i != arrayMS.length-1) System.out.print(", ");
        }
        System.out.println("}");
        int[] sorted = mergeSort(arrayMS);
        System.out.println("After Sorting: ");
        System.out.print("{ ");
        for(int i = 0; i<sorted.length; i++){
            System.out.print(sorted[i] + " ");
            if(i != sorted.length-1) System.out.print(", ");
        }
        System.out.println("}");
        System.out.println();
        System.out.println("----------------------------------");
        System.out.println();
    }

    public static int[] mergeSort(int[] array){
        if(array.length == 1) return array;
        int midIndex = array.length/2;
        int[] right = mergeSort(Arrays.copyOfRange(array, 0, midIndex));
        int[] left = mergeSort(Arrays.copyOfRange(array, midIndex, array.length));
        return merge(right, left);
    }

    public static int[] merge(int[] array1, int[] array2){
        int[] combined = new int[array1.length + array2.length];
        int index = 0;
        int i = 0;
        int j = 0;

        while(i < array1.length && j < array2.length){
            if(array1[i] < array2[j]){
                combined[index] = array1[i];
                index++;
                i++;
            }else{
                combined[index] = array2[j];
                index++;
                j++;
            }
        }

        while(i<array1.length){
            combined[index] = array1[i];
            index++;
            i++;
        }

        while(j<array2.length){
            combined[index] = array2[j];
            index++;
            j++;
        }
        return combined;
    }

}
