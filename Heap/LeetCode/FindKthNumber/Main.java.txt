public class Main {
    public static void main(String[] args) {
        Heap heap = new Heap();

         //testing to check if the smallest kth element is found in the heap
        System.out.println("Finding the smallest kth element in the heap: ");
        int[] nums3 = {9, 3, 2, 11, 7, 10, 4, 5};
        int k3 = 5;
        int ans = heap.findKthSmallest(nums3, k3);
        System.out.print("The given array is: ");
        System.out.print("{ ");
        for(int i=0; i<nums3.length; i++){
            System.out.print(nums3[i]);
            if(i != nums3.length-1) System.out.print(", ");
        }
        System.out.print(" }");
        System.out.println();
        System.out.println("The smallest " + k3 + "th element is: " + ans);
        System.out.println();

    }

}
