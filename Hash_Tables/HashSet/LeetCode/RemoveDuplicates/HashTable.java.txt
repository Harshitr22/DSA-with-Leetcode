public class HashTable {
    private int size = 7;
    private Node[] dataMap;

    class Node{
        String key;
        int value;
        Node next;

        Node(String key, int value){
            this.key = key;
            this.value = value;
        }
    }

    HashTable(){
        dataMap = new Node[size];
    }

    //Remove Duplicates
    public List<Integer> removeDuplicates(List<Integer> nums) {
        HashSet<Integer> set = new HashSet<>();
        for(int i: nums){
            set.add(nums.get(i));
        }
        return new ArrayList<>(set);
    }

}
