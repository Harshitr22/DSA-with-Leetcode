import java.util.*;

public class HashTable {
    private int size = 7;
    private Node[] dataMap;

    class Node{
        String key;
        int value;
        Node next;

        Node(String key, int value){
            this.key = key;
            this.value = value;
        }
    }

    HashTable(){
        dataMap = new Node[size];
    }


    //Sub Array sum, a problem we have to find a array in the given array to find if the array is alright
    public int[] subArray(int[] numbers, int target){
        HashMap<Integer, Integer> map = new HashMap<>();
        int[] indexes;
        int currentSum = 0;
        map.put(0, -1);
        for(int i=0; i<numbers.length; i++){
            currentSum += numbers[i];
            int desiredValue = currentSum - target;
            if(map.containsKey((desiredValue))){
                indexes = new int[]{map.get(desiredValue)+1, i};
                return indexes;
            }
            map.put(currentSum, i);
        }
        return null;
    }

}
