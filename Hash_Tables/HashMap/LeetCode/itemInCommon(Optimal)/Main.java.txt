        System.out.println("There are some items which are common in these arrays: " + itemInCommon(array0, array));

    }


    // This approach is the OPTIMAL APPROACH
    public static boolean itemInCommon(int[] array1, int[] array2){
        HashMap<Integer, Boolean> hashMap = new HashMap<>();

        for(int i : array1){
            hashMap.put(i, true);
        }

        for(int j : array2){
            if(hashMap.get(j) != null) return true;
        }
        return false;
    }

}
