import java.util.ArrayList;

public class HashTable {
    private int size = 7;
    private Node[] dataMap;

    class Node{
        String key;
        int value;
        Node next;

        Node(String key, int value){
            this.key = key;
            this.value = value;
        }
    }

    HashTable(){
        dataMap = new Node[size];
    }

    //Printing the hash table
    public void printTable(){
        for(int i=0; i< dataMap.length; i++){
            System.out.println(i + ": ");
            Node temp = dataMap[i];
            while(temp != null){
                System.out.println("  {" + temp.key + " : " + temp.value + "}");
                temp = temp.next;
            }
        }
    }

    //A method to formulate the Hash and calculate it to the index
    public int hash(String key){
        int hash = 0;
        char[] keyChars = key.toCharArray();
        for(int i=0; i<keyChars.length; i++){
            int asciiValue = keyChars[i];
            hash = (hash + asciiValue*23) % dataMap.length; // The reason we divide it by 7 is to establish the reasoning to keep it under the given size of an arraylist
        }
        return hash;
    }


    public ArrayList keys(){
        ArrayList<String> allKeys = new ArrayList<>();
        for(int i=0; i<dataMap.length; i++){
            Node temp = dataMap[i];
            while(temp != null){
                allKeys.add(temp.key);
                temp = temp.next;
            }
        }
        return allKeys;

    }

}
