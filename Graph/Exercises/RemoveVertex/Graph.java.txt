import java.util.ArrayList;
import java.util.HashMap;

public class Graph {
    private HashMap<String, ArrayList<String>> map = new HashMap<>();

    public void printGraph(){
        System.out.println(map);
    }

      //Adding the vertex in the graph
    public boolean addVertex(String vertex){
        if(map.get(vertex) == null){
            map.put(vertex, new ArrayList<>());
            return true;
        }
        return false;
    }

    //Adding edge to the vertex
    public boolean addEdge(String vertex1, String vertex2){
        if(map.get(vertex1) == null || map.get(vertex2) == null) return false;
        if(map.get(vertex1).contains(vertex2) || map.get(vertex2).contains(vertex1))  return false;
        map.get(vertex1).add(vertex2);
        map.get(vertex2).add(vertex1);
        return true;
    }

    //Removing the edge from the graph
    public boolean removeEdge(String vertex1, String vertex2){
        if(map.get(vertex1) == null || map.get(vertex2) == null) return false;
        if(map.get(vertex1).contains(vertex2) || map.get(vertex2).contains(vertex1))  return false;
        map.get(vertex1).remove(vertex2);
        map.get(vertex2).remove(vertex1);
        return true;
    }

    //Removing the vertex from the graph
    public boolean removeVertex(String vertex){
        if(map.get(vertex) != null){
            for(String otherVertex : map.get(vertex)){
                map.get(otherVertex).remove(vertex);
            }
            map.remove(vertex);
            return true;
        }
        return false;
    }

}
